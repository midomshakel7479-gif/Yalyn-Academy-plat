async function loadTeacherDashboard() { try { const res = await fetch(cfg.webAppUrl, { method: "POST", body: JSON.stringify({ action: "getTeacherDashboardData" }) }); const data = await res.json(); if(data.success) { globalStudents = data.students || []; globalResults = data.results || []; // جلب جميع الامتحانات المتاحة من خلفية السكريبت لتعبئة القوائم const examsRes = await fetch(cfg.webAppUrl, { method: "POST", body: JSON.stringify({ action: "getStudentDashboardData", grade: "", studentName: "" }) }); const examsData = await examsRes.json(); // تخزين الامتحانات العامة إن وجدت window.globalExamsList = examsData.exams || []; populateGradeDropdown(); toast('تم تحديث البيانات وجلب الصفوف والامتحانات بنجاح.'); } } catch (e) { console.error(e); toast('خطأ في جلب البيانات.'); } } function populateGradeDropdown() { const gradeSelect = document.getElementById('filterGrade'); const currentVal = gradeSelect.value; const gradesSet = new Set(); globalStudents.forEach(s => { if(s.grade) gradesSet.add(s.grade.toString().trim()); }); globalResults.forEach(r => { if(r.grade) gradesSet.add(r.grade.toString().trim()); }); gradeSelect.innerHTML = ``; gradesSet.forEach(grade => { gradeSelect.innerHTML += ``; }); if(gradesSet.has(currentVal)) gradeSelect.value = currentVal; onGradeSelected(); // تحديث قائمة الامتحانات بناءً على الصف الحالي } function onGradeSelected() { const selectedGrade = document.getElementById('filterGrade').value.trim(); const examSelect = document.getElementById('filterExamTitle'); examSelect.innerHTML = ``; if(!selectedGrade) { filterDashboardData(); return; } // جمع الامتحانات الخاصة بهذا الصف من النتائج المسجلة أو من قائمة الامتحانات العامة const examsSet = new Set(); // البحث في نتائج الطلاب السابقة لهذا الصف globalResults.forEach(r => { if(r.grade && r.grade.toString().trim() === selectedGrade && r.examTitle) { examsSet.add(r.examTitle.toString().trim()); } }); // البحث في قائمة الامتحانات العامة إن توفرت if(window.globalExamsList) { window.globalExamsList.forEach(ex => { examsSet.add(ex.examTitle.toString().trim()); }); } examsSet.forEach(ex => { examSelect.innerHTML += ``; }); filterDashboardData(); } function filterDashboardData() { const selectedGrade = document.getElementById('filterGrade').value.trim(); const selectedExam = document.getElementById('filterExamTitle').value.trim(); const submittedTable = document.getElementById('submittedStudentsTable'); const notSubmittedTable = document.getElementById('notSubmittedStudentsTable'); // حماية صارمة: لا يتم عرض أي نتائج أو غياب إلا إذا تم اختيار الصف AND اسم الامتحان معاً if(!selectedGrade || !selectedExam || selectedExam === "") { submittedTable.innerHTML = `الرجاء اختيار الصف الدراسي وعنوان الامتحان من القائمة بالأعلى`; notSubmittedTable.innerHTML = `الرجاء اختيار الصف الدراسي وعنوان الامتحان من القائمة بالأعلى`; document.getElementById('submittedCount').textContent = "0"; document.getElementById('notSubmittedCount').textContent = "0"; return; } const gradeStudents = globalStudents.filter(s => s.grade && s.grade.toString().trim() === selectedGrade); const filteredResults = globalResults.filter(r => { const matchGrade = r.grade && r.grade.toString().trim() === selectedGrade; const matchExam = r.examTitle && r.examTitle.toString().trim().toLowerCase() === selectedExam.toLowerCase(); return matchGrade && matchExam; }); submittedTable.innerHTML = ""; if(filteredResults.length === 0) { submittedTable.innerHTML = `لا توجد نتائج مسجلة لهذا الامتحان بالذات`; document.getElementById('submittedCount').textContent = "0"; } else { document.getElementById('submittedCount').textContent = filteredResults.length; filteredResults.forEach(r => { submittedTable.innerHTML += ` ${r.name} ${r.examTitle} ${r.score} ${r.date} `; }); } const submittedStudentNames = new Set(filteredResults.map(r => r.name.toString().trim().toLowerCase())); const notSubmittedStudents = gradeStudents.filter(s => !submittedStudentNames.has(s.name.toString().trim().toLowerCase())); notSubmittedTable.innerHTML = ""; if(notSubmittedStudents.length === 0) { notSubmittedTable.innerHTML = `ممتاز! جميع طلاب هذا الصف أتموا هذا الامتحان`; document.getElementById('notSubmittedCount').textContent = "0"; } else { document.getElementById('notSubmittedCount').textContent = notSubmittedStudents.length; notSubmittedStudents.forEach(s => { let cleanPhone = s.phone ? s.phone.toString().replace(/\D/g,'') : ''; if(cleanPhone.startsWith('0')) cleanPhone = '2' + cleanPhone; const waMsg = encodeURIComponent(`مرحباً ولي أمر الطالب/ة ${s.name}، نود تذكيركم بأن الطالب لم يقم بأداء امتحان (${selectedExam}) حتى الآن، برجاء سرعة الدخول للمنصة لأداء الاختبار.`); const waLink = cleanPhone ? `https://wa.me/${cleanPhone}?text=${waMsg}` : '#'; notSubmittedTable.innerHTML += ` ${s.name} ${s.code} ${s.phone ? `💬 ${s.phone}` : 'لا يوجد رقم'} `; }); } }
